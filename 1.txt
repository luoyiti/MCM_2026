  	extit{Output}: def compute_rank_ci(prob_matrix, alpha=0.05):
    """输入：prob_matrix (T x N) 为每位参赛者在每个可能名次上的概率分布。返回：每位参赛者的 (lower, upper) 95% 排名区间。方法：对边际分布累积并取分位数。"""
    import numpy as np
    T, N = prob_matrix.shape
    cdf = np.cumsum(prob_matrix, axis=1)
    lower = np.argmax(cdf >= alpha/2, axis=1) + 1
    upper = np.argmax(cdf >= 1-alpha/2, axis=1) + 1
    return list(zip(lower, upper))\\
